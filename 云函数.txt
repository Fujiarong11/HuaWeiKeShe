const cloudbase = require('@cloudbase/node-sdk');
const app = cloudbase.init({ env: cloudbase.TCB_ENV });
const db = app.database();
const _ = db.command;

exports.main = async (event) => {
  const { action, data } = event;

  // ========== 用户 user 模块 ==========
  if (action === "userRegister") {
    const exist = await db.collection("user").where({ id: data.id }).get();
    if (exist.data.length > 0) return { code: -1, msg: "用户已存在" };
    await db.collection("user").add({ data });
    return { code: 0, msg: "注册成功" };
  }
  if (action === "getUser") {
    const user = await db.collection("user").where({ id: data.id }).get();
    return { code: 0, data: user.data[0] };
  }

  // ========== 话题 topic 模块 ==========
  if (action === "addTopic") {
    await db.collection("topic").add({ data });
    return { code: 0, msg: "创建话题成功" };
  }
  if (action === "getAllTopic") {
    const list = await db.collection("topic").get();
    return { code: 0, data: list.data };
  }

  // ========== 提问 question 模块 ==========
  if (action === "publishQuestion") {
    await db.collection("question").add({ data });
    return { code: 0, msg: "发布提问成功" };
  }
  if (action === "getQuestionList") {
    const list = await db.collection("question")
      .where(data.filter || {})
      .skip(data.skip || 0)
      .limit(data.limit || 10)
      .get();
    return { code: 0, data: list.data };
  }

  // ========== 回答 answer 模块 ==========
  if (action === "publishAnswer") {
    await db.collection("answer").add({ data });
    // 提问回答数自增+1
    await db.collection("question")
      .where({ id: data.questionId })
      .update({ data: { answerCount: _.inc(1) } });
    return { code: 0, msg: "回答发布成功" };
  }
  if (action === "getAnswerByQid") {
    const list = await db.collection("answer").where({ questionId: data.qid }).get();
    return { code: 0, data: list.data };
  }

  // ========== 评论 comment 模块 ==========
  if (action === "publishComment") {
    await db.collection("comment").add({ data });
    // 回答评论数+1
    if (data.targetType === "answer") {
      await db.collection("answer")
        .where({ id: data.targetId })
        .update({ data: { commentCount: _.inc(1) } });
    }
    return { code: 0, msg: "评论发布成功" };
  }
  if (action === "getCommentByTid") {
    const list = await db.collection("comment").where({ targetId: data.tid }).get();
    return { code: 0, data: list.data };
  }

  // 未知操作
  return { code: -99, msg: "不存在该操作action" };
};

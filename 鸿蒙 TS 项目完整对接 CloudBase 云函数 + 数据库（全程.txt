import tcb from "../tcb";

/**
 * 用户登录
 * @param phone 手机号
 * @param password 密码
 */
export async function login(phone: string, password: string) {
  const res = await tcb.callFunction({
    name: "scfnodejshelloworld",
    data: {
      type: "login",
      phone,
      password
    }
  })
  return res.result;
}

/**
 * 用户注册
 * @param username 用户名
 * @param phone 手机号
 * @param password 密码
 */
export async function register(username: string, phone: string, password: string) {
  const res = await tcb.callFunction({
    name: "scfnodejshelloworld",
    data: {
      type: "register",
      username,
      phone,
      password
    }
  })
  return res.result;
}
